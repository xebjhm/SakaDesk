# D2b Blog Screens Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the inside of the blog sheet (recent feed, cards, reader, timeline rail, older/newer navigation) on the design-system tokens and primitives, with a tidier reading width and rhythm, without changing what the blog screens do.

**Architecture:** Behaviour is locked first (Playwright e2e on a pinned blog fixture plus vitest characterisation tests), on the unchanged code. A few tokens (`text-reading`, search-mark colours, `animate-reveal`), one date helper (`formatBlogDate`) and one breadcrumb (`blog_load`) are added, then each screen file is rewritten on tokens: inline stylesheets, glass, ambient orbs, hex-alpha group colours and the serif web font go. Logic (loading, caching, cancellation, navigation, translation, search highlighting, the rail's fisheye math) is kept as it is.

**Tech Stack:** React 18 + TypeScript, Tailwind 3.3 with design-system tokens, framer-motion 10 (`FadeOnChange`), lucide-react, vitest 4 + @testing-library/react + msw, Playwright, FastAPI + pydantic + structlog, pytest.

Spec: `docs/superpowers/specs/2026-11-05-d2b-blogs-design.md` (commit `0487e95`).

- Commit with `uv run git commit` (bare `git commit` fails: pre-commit is not on PATH). Conventional Commits, whole first line ≤ 72 chars, imperative, no trailing period, and end every message with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`. A hook enforces `docs/DOC_FORMATS.md`.

Width/height/position/grid-template/aspect arbitrary values and arbitrary variants (`min-[901px]:`, `[&_a]:`) are allowed. Spacing is a multiple of 4 px (no `*-0.5`, `*-1.5`, `*-2.5` padding, margin or gap). Overlays use `bg-scrim`.

### Adding i18n keys

Tasks 1 and 2 add keys. Use this one-liner from `frontend/` (the locale files round-trip byte-identically through `JSON.stringify(…, null, 2)`); pass a JSON object `{locale: {"section.key": "value"}}` as the argument and avoid apostrophes in values:

```bash
node -e '
const fs = require("fs");
const add = JSON.parse(process.argv[1]);
for (const [code, entries] of Object.entries(add)) {
  const file = `src/i18n/locales/${code}.json`;
  const data = JSON.parse(fs.readFileSync(file, "utf8"));
  for (const [dotted, value] of Object.entries(entries)) {
    const [section, key] = dotted.split(".");
    data[section] = { ...(data[section] ?? {}), [key]: value };
  }
  fs.writeFileSync(file, JSON.stringify(data, null, 2) + "\n");
}' '<JSON>'
```

`frontend/e2e/fixtures/blogs.ts`:

```ts
import type { Page, Route } from '@playwright/test'
import { buildBigRoom, pinBigRoom } from './bigRoom'

// Pinned blog data for the D2b locks and screenshots: two fictional members with CJK names,
// 42 + 6 posts with CJK + emoji titles and SVG pictures. Everything is answered at the network
// layer on top of pinBigRoom, which opens the app on the blog feature.

export const MEMBER_A = { id: '9901', name: '春野 ひなた' }
export const MEMBER_B = { id: '9902', name: '秋山 さくら' }
const DAY = 24 * 60 * 60 * 1000
/** Newest post: 2026-11-03 18:00 JST. */
const NEWEST = Date.UTC(2026, 10, 3, 9, 0, 0)

export interface FixturePost {
  id: string
  title: string
  published_at: string
  url: string
  thumbnail: string
  cached: boolean
}

/** A member's posts, newest first (index 0), as /api/blogs/list returns them. */
export function memberPosts(service: string, member: 'A' | 'B'): FixturePost[] {
  const spec =
    member === 'A'
      ? { count: 42, firstDay: 0, stepDays: 9, prefix: '1', label: '春の日記', emoji: '🌸' }
      : { count: 6, firstDay: 2, stepDays: 20, prefix: '2', label: '秋の便り', emoji: '🍁' }
  return Array.from({ length: spec.count }, (_, i) => {
    const n = String(spec.count - i).padStart(2, '0')
    const id = `${spec.prefix}${n}`
    return {
      id,
      title: `${spec.label} ${n} ${spec.emoji}`,
      published_at: new Date(NEWEST - (spec.firstDay + i * spec.stepDays) * DAY).toISOString(),
      url: `https://example.com/blog/${id}`,
      thumbnail: `/api/content/media/${service}/blogs/thumb-${id}.svg`,
      cached: true,
    }
  })
}

function contentHtml(service: string, post: FixturePost): string {
  const line = '春の風が吹いて、桜の花びらがゆっくりと舞っていました。'
  return [
    `<p>${post.title}の日です☀️<br>${line.repeat(3)}<br><br>`,
    'An English line to check how mixed text wraps at the reading width of the post.<br>',
    `<img src="/api/content/media/${service}/blogs/photo-${post.id}.svg"><br><br>`,
    '最後まで読んでくれてありがとう💐 <a href="https://example.com/more">もっと見る</a></p>',
  ].join('')
}

const svg = (w: number, h: number) =>
  `<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="${h}" viewBox="0 0 ${w} ${h}">` +
  `<rect width="${w}" height="${h}" fill="#cbd5e1"/><circle cx="${w / 2}" cy="${h * 0.4}" r="${w / 5}" fill="#94a3b8"/></svg>`

export interface PinBlogOptions {
  /** Service whose blogs are shown (default hinatazaka46). */
  service?: string
  /** What /api/blogs/recent answers (default ok). */
  recent?: 'ok' | 'empty' | 'error'
}

export async function pinBlogs(page: Page, opts: PinBlogOptions = {}): Promise<void> {
  const service = opts.service ?? 'hinatazaka46'
  await pinBigRoom(page, { messages: { current: buildBigRoom(10) }, service, feature: 'blogs' })

  const posts = { A: memberPosts(service, 'A'), B: memberPosts(service, 'B') }
  const all = [...posts.A, ...posts.B]
  const memberOf = (id: string) => (posts.A.some((p) => p.id === id) ? MEMBER_A : MEMBER_B)
  const recent = [...posts.A.slice(0, 3), ...posts.B]
    .sort((a, b) => b.published_at.localeCompare(a.published_at))
    .map((p) => ({
      id: p.id,
      title: p.title,
      published_at: p.published_at,
      url: p.url,
      thumbnail: p.thumbnail,
      member_id: memberOf(p.id).id,
      member_name: memberOf(p.id).name,
    }))

  // Routes registered later win, so these override pinBigRoom's catch-all media route.
  await page.route('**/api/blogs/recent*', (route) => {
    if (opts.recent === 'error') return route.fulfill({ status: 500, json: { detail: 'pinned failure' } })
    return route.fulfill({ json: { service, posts: opts.recent === 'empty' ? [] : recent } })
  })
  await page.route('**/api/blogs/sync*', (route) =>
    route.fulfill({
      json: { status: 'ok', service, total_members: 2, total_blogs: all.length, last_sync: '2026-11-04T03:00:00Z' },
    }),
  )
  await page.route('**/api/blogs/members-with-thumbnails*', (route) =>
    route.fulfill({
      json: { service, members: [{ ...MEMBER_A, thumbnail: 'a.jpg' }, { ...MEMBER_B, thumbnail: null }] },
    }),
  )
  await page.route('**/api/blogs/list*', (route) => {
    const memberId = new URL(route.request().url()).searchParams.get('member_id')
    const member = memberId === MEMBER_B.id ? MEMBER_B : MEMBER_A
    return route.fulfill({
      json: { member_id: member.id, member_name: member.name, blogs: member === MEMBER_A ? posts.A : posts.B },
    })
  })
  await page.route('**/api/blogs/content*', (route) => {
    const id = new URL(route.request().url()).searchParams.get('blog_id')
    const post = all.find((p) => p.id === id)
    if (!post) return route.fulfill({ status: 404, json: { detail: 'not found' } })
    return route.fulfill({
      json: {
        meta: { id: post.id, member_name: memberOf(post.id).name, title: post.title, published_at: post.published_at, url: post.url },
        content: { html: contentHtml(service, post) },
        images: [],
      },
    })
  })
  const image = (w: number, h: number) => (route: Route) =>
    route.fulfill({ body: svg(w, h), contentType: 'image/svg+xml' })
  await page.route('**/api/content/media/*/blogs/thumb-*', image(300, 400))
  await page.route('**/api/content/media/*/blogs/photo-*', image(600, 400))
  await page.route('**/api/blogs/member-thumbnail/**', image(200, 200))
  // The Hinatazaka mascot in the member dialog is a CDN picture.
  await page.route('https://cdn.hinatazaka46.com/**', image(200, 200))
}
```

```ts
/** Blog date styles: card and older/newer (short), rail tooltip (medium), reader (long). */
export type BlogDateStyle = 'short' | 'medium' | 'long';

const BLOG_DATE_OPTIONS: Record<BlogDateStyle, Intl.DateTimeFormatOptions> = {
  short: { month: 'short', day: 'numeric' },
  medium: { year: 'numeric', month: 'short', day: 'numeric' },
  long: { year: 'numeric', month: 'long', day: 'numeric', hour: '2-digit', minute: '2-digit', hourCycle: 'h23' },
};

/**
 * A blog date in the app language (pass i18n.language). '' for an invalid date; an
 * invalid locale tag falls back to the runtime default.
 */
export function formatBlogDate(date: Date, locale: string | undefined, style: BlogDateStyle): string {
  if (Number.isNaN(date.getTime())) return '';
  try {
    return new Intl.DateTimeFormat(locale, BLOG_DATE_OPTIONS[style]).format(date);
  } catch {
    return new Intl.DateTimeFormat(undefined, BLOG_DATE_OPTIONS[style]).format(date);
  }
}
```

```tsx
import type { RecentPost } from '../../../types';
```

```tsx
    const [imageLoaded, setImageLoaded] = useState(false);
    const [imageError, setImageError] = useState(false);

            onClick={onClick}

                        src={post.thumbnail}
                        alt=""
                        loading="lazy"
                        onLoad={() => setImageLoaded(true)}
                        onError={() => setImageError(true)}
                        className={cn(
                            'absolute inset-0 h-full w-full object-cover object-top',
                            'transition duration-base ease-standard group-hover:scale-105 motion-reduce:transform-none',
```

```tsx
// frontend/src/features/blogs/components/TimelineRail.tsx
// Pointer shortcut through a member's posts: top = newest, bottom = oldest, with a fisheye
// zoom after a dwell. Keyboard users have ← / → and the older/newer pills.
import React, { useState, useRef, useMemo, useEffect, useCallback } from 'react';
import type { BlogMeta } from '../../../types';
import { Card } from '../../../design/primitives';
import { cn } from '../../../utils/classnames';
import { formatBlogDate } from '../../../utils/dateFormatters';
import { useTranslation } from '../../../i18n';

interface TimelineRailProps {
    blogs: BlogMeta[];
    currentIndex: number;
    onSelect: (index: number) => void;
}

// Group blogs by year for section markers
interface YearGroup {
    year: number;
    startIndex: number;
    count: number;
}

// Fisheye zoom configuration
const ZOOM_CONFIG = {
    dwellTime: 800,
    dwellTolerance: 10,
    magnifiedWindow: 50,
    magnification: 5,
};

/** Space above the newest and below the oldest post, in px. */
const RAIL_PADDING = 30;

export const TimelineRail: React.FC<TimelineRailProps> = ({
    blogs,
    currentIndex,
    onSelect,
}) => {
    const { i18n } = useTranslation();
    const railRef = useRef<HTMLDivElement>(null);
    const [hoverInfo, setHoverInfo] = useState<{
        y: number;
        index: number;
    } | null>(null);

    // Zoom state
    const [isZoomed, setIsZoomed] = useState(false);
    const [zoomCenter, setZoomCenter] = useState<number | null>(null);
    const dwellTimerRef = useRef<NodeJS.Timeout | null>(null);
    const lastHoverIndexRef = useRef<number | null>(null);

    // Group blogs by year
    const yearGroups = useMemo((): YearGroup[] => {
        if (blogs.length === 0) return [];

        const groups: YearGroup[] = [];
        let currentYear = -1;

        blogs.forEach((blog, index) => {
            const year = new Date(blog.published_at).getFullYear();
            if (year !== currentYear) {
                if (groups.length > 0) {
                    groups[groups.length - 1].count = index - groups[groups.length - 1].startIndex;
                }
                groups.push({ year, startIndex: index, count: 0 });
                currentYear = year;
            }
        });

        if (groups.length > 0) {
            groups[groups.length - 1].count = blogs.length - groups[groups.length - 1].startIndex;
        }

        return groups;
    }, [blogs]);

    // Fisheye mapping: index → normalized position (0-1)
    const indexToNormalizedPosition = useCallback((index: number): number => {
        if (blogs.length <= 1) return 0;
        const normalizedIndex = index / (blogs.length - 1);

        if (!isZoomed || zoomCenter === null) {
            return normalizedIndex;
        }

        const center = zoomCenter / (blogs.length - 1);
        // Clamp windowSize so magnified region doesn't exceed available space
        const rawWindowSize = ZOOM_CONFIG.magnifiedWindow / (blogs.length - 1);
        const mag = ZOOM_CONFIG.magnification;
        const windowSize = Math.min(rawWindowSize, 0.45 / mag);

        const dist = normalizedIndex - center;
        const absDist = Math.abs(dist);

        if (absDist <= windowSize) {
            const t = absDist / windowSize;
            const eased = t * t * (3 - 2 * t);
            const expandedDist = dist * (1 + (mag - 1) * (1 - eased));
            return Math.max(0, Math.min(1, center + expandedDist));
        } else {
            const magnifiedSpace = windowSize * mag;
            const remainingSpace = Math.max((1 - magnifiedSpace * 2) / 2, 0.01);
            const normalRemaining = Math.max(0.5 - windowSize, 0.001);

            if (dist > 0) {
                const outsideDist = absDist - windowSize;
                const compressedDist = outsideDist * (remainingSpace / normalRemaining);
                return Math.max(0, Math.min(1, center + magnifiedSpace + compressedDist));
            } else {
                const outsideDist = absDist - windowSize;
                const compressedDist = outsideDist * (remainingSpace / normalRemaining);
                return Math.max(0, Math.min(1, center - magnifiedSpace - compressedDist));
            }
        }
    }, [blogs.length, isZoomed, zoomCenter]);

    // Inverse fisheye mapping: normalized position (0-1) → index
    const normalizedPositionToIndex = useCallback((pos: number): number => {
        if (blogs.length <= 1) return 0;

        if (!isZoomed || zoomCenter === null) {
            return Math.round(pos * (blogs.length - 1));
        }

        const center = zoomCenter / (blogs.length - 1);
        const rawWindowSize = ZOOM_CONFIG.magnifiedWindow / (blogs.length - 1);
        const mag = ZOOM_CONFIG.magnification;
        const windowSize = Math.min(rawWindowSize, 0.45 / mag);
        const magnifiedSpace = windowSize * mag;

        const dist = pos - center;
        const absDist = Math.abs(dist);

        let normalizedIndex: number;

        if (absDist <= magnifiedSpace) {
            const sign = dist >= 0 ? 1 : -1;
            const originalDist = (absDist / mag);
            normalizedIndex = center + sign * originalDist;
        } else {
            const remainingSpace = Math.max((1 - magnifiedSpace * 2) / 2, 0.01);
            const normalRemaining = Math.max(0.5 - windowSize, 0.001);

            if (dist > 0) {
                const compressedDist = absDist - magnifiedSpace;
                const originalDist = compressedDist * (normalRemaining / remainingSpace);
                normalizedIndex = center + windowSize + originalDist;
            } else {
                const compressedDist = absDist - magnifiedSpace;
                const originalDist = compressedDist * (normalRemaining / remainingSpace);
                normalizedIndex = center - windowSize - originalDist;
            }
        }

        return Math.round(Math.max(0, Math.min(1, normalizedIndex)) * (blogs.length - 1));
    }, [blogs.length, isZoomed, zoomCenter]);

    // Clear dwell timer on cleanup
    useEffect(() => {
        return () => {
            if (dwellTimerRef.current) {
                clearTimeout(dwellTimerRef.current);
            }
        };
    }, []);

    // Handle mouse move
    const handleMouseMove = (e: React.MouseEvent) => {
        if (!railRef.current || blogs.length === 0) return;
        const rect = railRef.current.getBoundingClientRect();
        const y = e.clientY - rect.top;
        const usableHeight = rect.height - RAIL_PADDING * 2;

        const normalizedY = Math.max(0, Math.min(1, (y - RAIL_PADDING) / usableHeight));
        const index = normalizedPositionToIndex(normalizedY);
        const clampedIndex = Math.max(0, Math.min(blogs.length - 1, index));

        setHoverInfo({ y, index: clampedIndex });

        // Dwell detection for zoom activation
        if (!isZoomed) {
            const lastIndex = lastHoverIndexRef.current;
            const indexChanged = lastIndex === null ||
                Math.abs(clampedIndex - lastIndex) > ZOOM_CONFIG.dwellTolerance;

            if (indexChanged) {
                if (dwellTimerRef.current) {
                    clearTimeout(dwellTimerRef.current);
                }
                lastHoverIndexRef.current = clampedIndex;

                dwellTimerRef.current = setTimeout(() => {
                    setZoomCenter(clampedIndex);
                    setIsZoomed(true);
                }, ZOOM_CONFIG.dwellTime);
            }
        } else {
            if (zoomCenter !== null && Math.abs(clampedIndex - zoomCenter) > ZOOM_CONFIG.dwellTolerance * 2) {
                setZoomCenter(clampedIndex);
            }
        }
    };

    const handleMouseLeave = () => {
        if (dwellTimerRef.current) {
            clearTimeout(dwellTimerRef.current);
            dwellTimerRef.current = null;
        }
        lastHoverIndexRef.current = null;
        setIsZoomed(false);
        setZoomCenter(null);
        setHoverInfo(null);
    };

    const handleClick = () => {
        if (hoverInfo) {
            onSelect(hoverInfo.index);
        }
    };

    if (blogs.length === 0) return null;

    const getYFromIndex = (index: number, ref: HTMLDivElement | null) => {
        if (!ref) return RAIL_PADDING;
        const usableHeight = ref.clientHeight - RAIL_PADDING * 2;
        const normalizedPos = indexToNormalizedPosition(index);
        return RAIL_PADDING + normalizedPos * usableHeight;
    };

    const hovered = hoverInfo ? blogs[hoverInfo.index] : undefined;

    return (
        <div
            ref={railRef}
            data-testid="timeline-rail"
            aria-hidden="true"
            className="absolute inset-x-0 top-1/2 flex h-[70%] min-h-[300px] -translate-y-1/2 cursor-pointer select-none flex-col items-center"
            onMouseMove={handleMouseMove}
            onMouseLeave={handleMouseLeave}
            onClick={handleClick}
        >
            {/* Track */}
            <div
                className="absolute left-1/2 -translate-x-1/2 border-l-2 border-subtle"
                style={{ top: RAIL_PADDING, bottom: RAIL_PADDING }}
            />

            {/* Year markers (a label is skipped when it would overlap the previous one) */}
            {yearGroups.map((group, idx) => {
                const position = getYFromIndex(group.startIndex, railRef.current);
                const prevGroup = yearGroups[idx - 1];
                const prevPosition = prevGroup
                    ? getYFromIndex(prevGroup.startIndex, railRef.current)
                    : -100;

                if (idx > 0 && position - prevPosition < 24) return null;

                return (
                    <div
                        key={group.year}
                        className="absolute right-full mr-2 flex -translate-y-1/2 items-center transition-[top] duration-base ease-standard"
                        style={{ top: position }}
                    >
                        <span className="whitespace-nowrap text-caption font-medium text-muted">{group.year}</span>
                        <div className="ml-2 w-1.5 border-t border-strong" />
                    </div>
                );
            })}

            {/* Current post */}
            <div
                className="absolute left-1/2 h-2.5 w-2.5 -translate-x-1/2 -translate-y-1/2 rounded-pill bg-group-accent shadow-group-glow transition-[top] duration-base ease-standard"
                style={{ top: getYFromIndex(currentIndex, railRef.current) }}
            />

            {/* Hover ring and card */}
            {hoverInfo && hovered && (
                <>
                    <div
                        className={cn(
                            'pointer-events-none absolute left-1/2 h-3.5 w-3.5 -translate-x-1/2 -translate-y-1/2 rounded-pill border-2 border-strong',
                            'transition-opacity duration-base ease-standard',
                            isZoomed ? 'opacity-90' : 'opacity-60',
                        )}
                        style={{ top: hoverInfo.y }}
                    />

                    <Card
                        className="pointer-events-none absolute right-full z-50 mr-4 min-w-[180px] max-w-[240px] -translate-y-1/2 overflow-hidden"
                        style={{
                            top: Math.min(
                                Math.max(hoverInfo.y, 180),
                                (railRef.current?.clientHeight || 400) - 180
                            ),
                        }}
                    >
                        {hovered.thumbnail && (
                            <div
                                className="aspect-[3/4] w-full bg-sunken bg-cover bg-center"
                                style={{ backgroundImage: `url(${hovered.thumbnail})` }}
                            />
                        )}
                        <div className="flex flex-col gap-1 p-3">
                            <div className="line-clamp-2 text-small font-semibold text-primary">{hovered.title}</div>
                            <div className="flex items-center justify-between gap-2 text-caption tabular-nums text-muted">
                                <span>{formatBlogDate(new Date(hovered.published_at), i18n.language, 'medium')}</span>
                                <span>#{blogs.length - hoverInfo.index}</span>
                            </div>
                        </div>
                    </Card>
                </>
            )}

            {/* Position, counted from the oldest post */}
            <div className="absolute bottom-1 left-1/2 -translate-x-1/2 whitespace-nowrap text-caption font-medium tabular-nums text-muted">
                {blogs.length - currentIndex}/{blogs.length}
            </div>
        </div>
    );
};
```

```tsx
// frontend/src/features/blogs/components/BlogNavFooter.tsx
// Older / newer post as two floating pills over the bottom of the post (← / → work too).
import React, { useEffect } from 'react';
import { ChevronLeft, ChevronRight } from 'lucide-react';
import type { BlogMeta } from '../../../types';
import { cn } from '../../../utils/classnames';
import { formatBlogDate } from '../../../utils/dateFormatters';
import { useTranslation } from '../../../i18n';

interface BlogNavFooterProps {
    prevBlog: BlogMeta | null;
    nextBlog: BlogMeta | null;
    onPrev: () => void;
    onNext: () => void;
}

const SIDE = cn(
    'pointer-events-auto group flex max-w-[45%] items-center gap-2 rounded-pill bg-pill py-2 shadow-pill',
    'transition-colors duration-fast ease-standard hover:bg-sunken',
    'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-group',
);
const CHEVRON = 'h-4 w-4 shrink-0 text-muted transition-colors duration-fast ease-standard group-hover:text-primary';

const truncateTitle = (title: string, maxLen: number = 18) =>
    title.length <= maxLen ? title : title.slice(0, maxLen) + '...';

export const BlogNavFooter: React.FC<BlogNavFooterProps> = ({
    prevBlog,
    nextBlog,
    onPrev,
    onNext,
}) => {
    const { t, i18n } = useTranslation();

    // Keyboard navigation
    useEffect(() => {
        const handleKeyDown = (e: KeyboardEvent) => {
            if (e.key === 'ArrowLeft' && prevBlog) {
                onPrev();
            } else if (e.key === 'ArrowRight' && nextBlog) {
                onNext();
            }
        };

        window.addEventListener('keydown', handleKeyDown);
        return () => window.removeEventListener('keydown', handleKeyDown);
    }, [prevBlog, nextBlog, onPrev, onNext]);

    // Don't render if no navigation available
    if (!prevBlog && !nextBlog) return null;

    const label = (blog: BlogMeta) => (
        <span className="min-w-0">
            <span className="block max-w-[160px] truncate text-small font-semibold text-primary">
                {truncateTitle(blog.title)}
            </span>
            <span className="block text-caption text-muted">
                {formatBlogDate(new Date(blog.published_at), i18n.language, 'short')}
            </span>
        </span>
    );

    return (
        <nav
            aria-label={t('blogs.postNavigation')}
            className="pointer-events-none absolute inset-x-6 bottom-4 z-10 flex items-center justify-between gap-4"
        >
            {prevBlog ? (
                <button type="button" onClick={onPrev} className={cn(SIDE, 'pl-3 pr-4 text-left')}>
                    <ChevronLeft aria-hidden="true" className={CHEVRON} />
                    <span className="sr-only">{t('blogs.olderPost')}: </span>
                    {label(prevBlog)}
                </button>
            ) : (
                <span />
            )}

            {nextBlog ? (
                <button type="button" onClick={onNext} className={cn(SIDE, 'pl-4 pr-3 text-right')}>
                    <span className="sr-only">{t('blogs.newerPost')}: </span>
                    {label(nextBlog)}
                    <ChevronRight aria-hidden="true" className={CHEVRON} />
                </button>
            ) : (
                <span />
            )}
        </nav>
    );
};
```
