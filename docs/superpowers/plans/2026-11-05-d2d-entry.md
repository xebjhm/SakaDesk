# D2d Entry Screens Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Put the entry screens of SakaDesk (ToS, landing, login, setup wizard, sync dialog, inline sync view, boot screen and the App error toast) on the design-system tokens and primitives without changing what they do, and lift a ringing call above dialogs that were open when it started.

**Architecture:** Behaviour is locked first (vitest characterisation tests on the components, Playwright on a fully mocked entry flow, with every `/api/auth/**` request answered by the test) on the unchanged code, and before screenshots are taken. Then a small foundation lands (a `bg-window` token, `bg-status-success`, `groupScopeStyle()` to show one service's colours on a surface, neutral `default` group tokens before a group is chosen), and each screen is rewritten on it: a shared `SyncProgressPanel` for the sync dialog and the inline sync view, the ToS / setup / sync dialogs on `Modal`'s standard layout, login with the official banner, the landing page as canvas + sheet + selectable cards. Last, the call overlay gets an interleaved layer above earlier dialogs.

**Tech Stack:** React 18 + TypeScript, Tailwind 3.3 with design-system tokens, tailwind-merge 2, lucide-react, vitest 4 + @testing-library/react + user-event + msw, Playwright, FastAPI + pydantic + structlog, pytest.

Spec: `docs/superpowers/specs/2026-11-05-d2d-entry-design.md` (decision numbers below refer to its Decisions table).

### Adding or removing i18n keys

From `frontend/` (the locale files round-trip byte-identically through `JSON.stringify(…, null, 2)`); pass a JSON object `{locale: {"section.key": "value"}}` and avoid apostrophes in values:

```bash
node -e '
const fs = require("fs");
const add = JSON.parse(process.argv[1]);
for (const [code, entries] of Object.entries(add)) {
  const file = `src/i18n/locales/${code}.json`;
  const data = JSON.parse(fs.readFileSync(file, "utf8"));
  for (const [dotted, value] of Object.entries(entries)) {
    const [section, key] = dotted.split(".");
    data[section] = { ...(data[section] ?? {}), [key]: value };
  }
  fs.writeFileSync(file, JSON.stringify(data, null, 2) + "\n");
}' '<JSON>'
```

```tsx
/** Answers the login calls; never a real network request (msw rejects unhandled ones). */
function pinLogin(opts: { loginStatus?: number; geo?: 'blocked' | 'open' | 'error' } = {}) {
    const calls = { login: [] as string[], init: [] as string[], geo: 0 };
    server.use(
        http.post('/api/auth/login', ({ request }) => {
            calls.login.push(new URL(request.url).searchParams.get('service') ?? '');
            return opts.loginStatus && opts.loginStatus !== 200
                ? new HttpResponse(null, { status: opts.loginStatus })
                : HttpResponse.json({ success: true });
        }),
        http.post('/api/settings/service/:service/init', ({ params }) => {
            calls.init.push(String(params.service));
            return HttpResponse.json({ ok: true });
        }),
        http.get('/api/auth/geo-availability', () => {
            calls.geo += 1;
            if (opts.geo === 'error') return HttpResponse.error();
            return HttpResponse.json({ blocked: opts.geo === 'blocked' });
        }),
    );
    return calls;
}
```

```ts
import type { Page } from '@playwright/test'
import { pinBlogs } from './blogs'

// The entry flow on mocked APIs: ToS, landing, login, setup and the first sync.
// Login tokens are single-session: EVERY /api/auth/** request is answered here, and any
// auth path this file does not know is aborted, so no test can reach the real login.

export const ALL_SERVICES = ['nogizaka46', 'sakurazaka46', 'hinatazaka46', 'yodel'] as const

export const RUNNING = {
  state: 'running', phase: 'scanning', phase_name: 'Scanning', phase_number: 1, completed: 5, total: 20,
  elapsed_seconds: 30, eta_seconds: 90, speed: 2.5, speed_unit: 'groups', detail: 'Fixture detail',
}
export const MEDIA = { ...RUNNING, phase: 'downloading', phase_name: 'Downloading', phase_number: 3, completed: 100, total: 500, speed_unit: 'files' }
export const COMPLETE = {
  state: 'complete', phase: 'complete', phase_number: 3, completed: 20, total: 20,
  elapsed_seconds: 60, speed: 2.5, speed_unit: 'groups',
}

export interface EntryState {
  /** Services /api/auth/status reports as logged in; a successful login adds its service. */
  authed: Set<string>
  /** Status POST /api/auth/login answers (200, or e.g. 500 for a failed login). */
  loginStatus: number
  /** What /api/auth/geo-availability answers for Yodel. */
  geoBlocked: boolean
  /** Services a login was POSTed for, in order. */
  loginCalls: string[]
  settings: Record<string, unknown>
  /** Bodies of POST /api/settings, in order. */
  settingsSaves: Record<string, unknown>[]
  /** Status POST /api/settings answers (500 shows the App error toast). */
  settingsSaveStatus: number
  /** /api/sync/progress per service (idle when missing); tests change it to move a sync on. */
  progress: Record<string, Record<string, unknown>>
}

export interface EntryOptions {
  /** ToS already accepted (default false: first launch). */
  tosAccepted?: boolean
  /** Services selected at start (default none: the landing page). */
  services?: string[]
  /** Output folder already set (default false: the setup wizard shows). */
  configured?: boolean
  /** Services logged in at start. */
  authed?: string[]
}

export async function pinEntry(page: Page, opts: EntryOptions = {}): Promise<EntryState> {
  // Content, blog, media and conversation routes; registered first so the routes below win.
  await pinBlogs(page)
  const services = opts.services ?? []
  const state: EntryState = {
    authed: new Set(opts.authed ?? []),
    loginStatus: 200,
    geoBlocked: false,
    loginCalls: [],
    settings: {
      output_dir: opts.configured ? '/fixture/out' : '',
      auto_sync_enabled: false,
      adaptive_sync_enabled: false,
      sync_interval_minutes: 15,
      is_configured: opts.configured ?? false,
      incoming_call_enabled: false,
      notifications_enabled: false,
      close_to_tray: true,
    },
    settingsSaves: [],
    settingsSaveStatus: 200,
    progress: {},
  }

  await page.route('**/api/app-state/prefs', (route) => {
    if (route.request().method() !== 'GET') return route.fulfill({ status: 204 })
    return route.fulfill({
      json: {
        language: 'en',
        ...(opts.tosAccepted ? { tos_accepted_at: '2026-01-01T00:00:00Z' } : {}),
        'sakadesk-app-state': JSON.stringify({
          state: {
            selectedServices: services,
            activeService: services[0] ?? null,
            activeFeatures: Object.fromEntries(services.map((s) => [s, 'messages'])),
          },
          version: 4,
        }),
      },
    })
  })

  // Auth: the catch-all first (later routes win), then the paths the app uses.
  await page.route('**/api/auth/**', (route) => route.abort('blockedbyclient'))
  await page.route((url) => url.pathname === '/api/auth/status', (route) =>
    route.fulfill({
      json: { services: Object.fromEntries(ALL_SERVICES.map((s) => [s, { authenticated: state.authed.has(s) }])) },
    }),
  )
  await page.route((url) => url.pathname === '/api/auth/login', (route) => {
    const service = new URL(route.request().url()).searchParams.get('service') ?? ''
    state.loginCalls.push(service)
    if (state.loginStatus !== 200) return route.fulfill({ status: state.loginStatus, json: { detail: 'pinned failure' } })
    state.authed.add(service)
    return route.fulfill({ json: { success: true } })
  })
  await page.route((url) => url.pathname === '/api/auth/geo-availability', (route) =>
    route.fulfill({ json: { blocked: state.geoBlocked } }),
  )
  await page.route((url) => url.pathname === '/api/auth/refresh-if-needed', (route) =>
    route.fulfill({ json: { refreshed: false } }),
  )

  await page.route((url) => url.pathname === '/api/settings', (route) => {
    if (route.request().method() === 'POST') {
      const body = route.request().postDataJSON() as Record<string, unknown>
      state.settingsSaves.push(body)
      if (state.settingsSaveStatus !== 200) return route.fulfill({ status: state.settingsSaveStatus, json: { detail: 'pinned failure' } })
      Object.assign(state.settings, body)
      if (typeof body.output_dir === 'string' && body.output_dir) state.settings.is_configured = true
    }
    return route.fulfill({ json: state.settings })
  })
  await page.route((url) => url.pathname.startsWith('/api/settings/service/'), (route) => route.fulfill({ json: {} }))
  await page.route((url) => url.pathname === '/api/settings/fresh', (route) => route.fulfill({ json: { is_fresh: false } }))

  await page.route('**/api/sync/start*', (route) => route.fulfill({ json: { status: 'started' } }))
  await page.route('**/api/sync/cancel*', (route) => route.fulfill({ json: { status: 'cancelled' } }))
  await page.route('**/api/sync/progress*', (route) => {
    const service = new URL(route.request().url()).searchParams.get('service') ?? ''
    return route.fulfill({ json: state.progress[service] ?? { state: 'idle' } })
  })
  await page.route('**/api/profile**', (route) => route.fulfill({ json: {} }))
  return state
}
```

```tsx
import React from 'react';

import { formatSyncTime, formatSyncSpeed, getSyncDetail } from '../../utils/syncFormatters';

const PHASE_ORDER = ['scanning', 'discovering', 'syncing', 'downloading', 'complete'];

export interface SyncProgressPanelProps {
    progress: SyncProgress;
}

/**
 * The shared body of the sync dialog and the inline sync view
 * (docs/superpowers/specs/2026-11-05-d2d-entry-design.md, decisions 9–11): count and percent,
 * the progress bar, three stat tiles, the detail row while running, and the phase steps.
 */
export const SyncProgressPanel: React.FC<SyncProgressPanelProps> = ({ progress }) => {
    const { t } = useTranslation();
    const total = progress.total ?? 0;
    const ratio = total > 0 ? (progress.completed || 0) / total : 0;
    const percent = Math.round(ratio * 100);
    const currentIndex = PHASE_ORDER.indexOf(progress.phase || '');
    const stats = [
        { key: 'elapsed', label: t('time.elapsed'), value: formatSyncTime(progress.elapsed_seconds) },
        { key: 'eta', label: t('time.eta'), value: progress.eta_seconds ? formatSyncTime(progress.eta_seconds) : '--:--' },
        { key: 'speed', label: t('time.speed'), value: formatSyncSpeed(progress.speed, progress.speed_unit || 'it') },
    ];
    const steps = [
        { phase: 'scanning', label: t('sync.scan') },
        { phase: 'syncing', label: t('sync.syncing') },
        { phase: 'downloading', label: t('sync.download') },
    ];

    return (
        <div className="flex flex-col gap-5">
            <div className="flex flex-col gap-2">
                <div className="flex justify-between text-small font-semibold tabular-nums">
                    <span className="text-secondary">
                        {progress.completed?.toLocaleString() || 0} / {progress.total?.toLocaleString() || 0}
                    </span>
                    <span className="text-primary">{`${percent}%`}</span>
                </div>
                <div
                    role="progressbar"
                    aria-label={t('sync.progress')}
                    aria-valuemin={0}
                    aria-valuemax={100}
                    aria-valuenow={percent}
                    className="h-2 overflow-hidden rounded-pill bg-sunken"
                >
                    <div
                        className="h-full rounded-pill bg-group-accent transition-all duration-base ease-standard motion-reduce:transition-none"
                        style={{ width: `${ratio * 100}%` }}
                    />
                </div>
            </div>

            <div className="grid grid-cols-3 gap-4">
                {stats.map((s) => (
                    <div key={s.key} className="flex flex-col gap-1 rounded-card bg-sunken p-3 text-center">
                        <span className="text-caption text-secondary">{s.label}</span>
                        <span className="font-mono text-title font-semibold tabular-nums text-primary">{s.value}</span>
                    </div>
                ))}
            </div>

            {/* Only while running: on completion the backend leaves phase_number at 3, and the
                "do not close" line (and the spinner) would make a finished sync look stuck. */}
            {progress.state === 'running' && (progress.phase_number === 3 ? (
                <div className="flex items-center justify-center gap-2 rounded-card bg-sunken px-4 py-3">
                    <Loader2 aria-hidden="true" className="h-4 w-4 shrink-0 animate-spin text-status-warning motion-reduce:animate-none" />
                    <span className="text-small font-semibold text-primary">{t('sync.downloadingMedia')}</span>
                </div>
            ) : (

                    <Loader2 aria-hidden="true" className="h-4 w-4 shrink-0 animate-spin text-secondary motion-reduce:animate-none" />
                    <span className="truncate text-small font-semibold text-primary">
                        {getSyncDetail(progress, t, t('sync.processing'))}
                        {progress.detail_extra && ` ${progress.detail_extra}`}
                    </span>
                </div>
            ))}

            <ol className="flex justify-center gap-6">
                {steps.map((step) => {
                    const isActive = progress.phase === step.phase || (step.phase === 'scanning' && progress.phase === 'discovering');
                    const isComplete = currentIndex > PHASE_ORDER.indexOf(step.phase);
                    return (
                        <li key={step.phase} aria-current={isActive ? 'step' : undefined} className="flex flex-col items-center gap-1">
                            <span
                                aria-hidden="true"
                                className={cn(
                                    'h-3 w-3 rounded-pill transition-colors duration-fast ease-standard',

                                )}
                            />
                            <span className={cn('text-caption', isActive ? 'font-semibold text-group-strong' : 'text-muted')}>{step.label}</span>
                        </li>
                    );
                })}
            </ol>
        </div>
    );
};
```

```tsx
    const isLight = groupTokens[groupId].official.headerStyle === 'light';

    const doLogin = async () => {
        setIsLoading(true);
        setError(null);
        try {
            const res = await fetch(
                `/api/auth/login?service=${encodeURIComponent(serviceId)}`,
                { method: 'POST' }
            );
            if (!res.ok) throw new Error(t('login.loginFailed'));

            // Initialize service settings entry
            await fetch(
                `/api/settings/service/${encodeURIComponent(serviceId)}/init`,
                { method: 'POST' }
            );

            // Brief delay before advancing to prevent concurrent browser launches
            await new Promise(resolve => setTimeout(resolve, 1500));
            logDialogLoad('login', 'ok');
            onSuccess();
        } catch (err: unknown) {
            logDialogLoad('login', 'error', undefined, 'failed');
            setError(err instanceof Error ? err.message : t('login.loginFailed'));
        } finally {
            setIsLoading(false);
        }
    };

    const handleLogin = async () => {
        // Yodel's web service is Japan-only. Warn before a login attempt from
        // another region, since the login/data servers reject non-JP connections.
        if (serviceId === 'yodel') {
            try {
                const res = await fetch('/api/auth/geo-availability?service=yodel', { cache: 'no-store' });
                if (!res.ok) {
                    logDialogLoad('login_geo', 'error', undefined, 'failed');
                } else if ((await res.json()).blocked) {
                    logDialogLoad('login_geo', 'skipped', undefined, 'blocked');
                    setShowGeoWarning(true);
                    return;
                } else {
                    logDialogLoad('login_geo', 'ok');
                }
            } catch {
                // Check failed (offline, etc.) — don't block the user; let login proceed.
                logDialogLoad('login_geo', 'error', undefined, 'unreachable');
            }
        }
        await doLogin();
    };

    const title = isFreshPrompt
        ? t('login.connectPromptTitle')
        : isDisconnected
            ? t('login.sessionExpired')
            : t('login.loginRequired');
    const description = isFreshPrompt
        ? t('login.connectPromptDesc', { service: service?.displayName ?? serviceId })
        : isDisconnected
            ? t('login.sessionExpiredDesc', {
                service: service?.displayName ?? serviceId,
                feature: feature ? t(feature.labelKey) : featureId,
            })
            : t('login.premiumFeatureDesc', {
                feature: feature ? t(feature.labelKey) : featureId,
                service: service?.name ?? serviceId,
            });

    return (
        <>
            <Modal isOpen onClose={onClose} layout="bare" labelledBy={titleId} logId="login" className="max-w-md">
                <div className="flex min-h-0 flex-col" style={groupScopeStyle(groupId)}>
                    {/* The service's official banner, drawn like its chat header. */}
                    <div
                        data-testid="login-banner"
                        className={cn('flex shrink-0 items-center gap-3 px-6 py-4', isLight ? 'bg-sheet text-group-title' : 'bg-group-title text-inverse')}
                    >
                        <span
                            aria-hidden="true"
                            className={cn('flex h-10 w-10 shrink-0 items-center justify-center rounded-pill', isLight ? 'bg-group-subtle' : 'bg-viewer-control')}
                        >
                            <LogIn className="h-5 w-5" />
                        </span>
                        <div className="min-w-0 flex-1">
                            <h2 id={titleId} className="truncate text-title font-semibold">{title}</h2>
                            <p className="truncate text-small">{service?.displayName ?? serviceId}</p>
                        </div>
                        <IconButton
                            size="sm"
                            variant={isLight ? 'ghost' : 'overlay'}
                            aria-label={t('common.close')}
                            icon={<X />}
                            onClick={onClose}
                        />
                    </div>
                    {isLight && <div className="h-1 shrink-0 bg-group-title-bar" />}

                    <div className="flex flex-col gap-4 overflow-y-auto p-6">
                        <p className="text-body text-secondary">{description}</p>

                        {error && (
                            <p role="alert" className="flex items-center gap-2 text-small text-status-danger">
                                <AlertCircle aria-hidden="true" className="h-4 w-4 shrink-0" />
                                {error}
                            </p>
                        )}

                        <div className="flex gap-3">
                            <Button variant="secondary" className="flex-1" onClick={onClose} disabled={isLoading}>
                                {isFreshPrompt ? t('login.maybeLater') : t('common.cancel')}
                            </Button>
                            <Button variant="primary" className="flex-1" onClick={handleLogin} disabled={isLoading}>
                                {isLoading ? (
                                    <>
                                        <Loader2 aria-hidden="true" className="h-4 w-4 animate-spin motion-reduce:animate-none" />
                                        {t('login.connecting')}
                                    </>
                                ) : (
                                    isFreshPrompt ? t('login.connectAccount') : t('common.login')
                                )}
                            </Button>
                        </div>

                        <p className="text-center text-caption text-muted">{t('login.credentialsSaved')}</p>
                    </div>
                </div>
            </Modal>
            <ConfirmDialog
                open={showGeoWarning}
                title={t('login.yodelJpOnlyTitle')}
                message={t('login.yodelJpOnlyMessage')}
                confirmLabel={t('login.loginAnyway')}
                cancelLabel={t('common.close')}
                variant="warning"
                onConfirm={() => { setShowGeoWarning(false); doLogin(); }}
                onCancel={() => { logDialogLoad('login', 'cancelled'); setShowGeoWarning(false); }}
            />
        </>
    );
};
```

```tsx
/**
```

```ts
export function interleavedZIndex(order: number): number {
```

```ts
// frontend/src/features/incomingCall/useAnchorRect.ts
import { useLayoutEffect, useState, type RefObject } from 'react';

export interface AnchorRect {
    left: number;
    top: number;
    width: number;
    height: number;
}

/** The viewport rectangle of `ref` while `enabled`, kept current on resize (where a lifted call is drawn). */
export function useAnchorRect(ref: RefObject<HTMLElement>, enabled: boolean): AnchorRect | null {
    const [rect, setRect] = useState<AnchorRect | null>(null);
    useLayoutEffect(() => {
        const el = ref.current;
        if (!enabled || !el) return;
        const measure = () => {
            const r = el.getBoundingClientRect();
            setRect((prev) => (
                prev && prev.left === r.left && prev.top === r.top && prev.width === r.width && prev.height === r.height
                    ? prev
                    : { left: r.left, top: r.top, width: r.width, height: r.height }
            ));
        };
        measure();
        const observer = typeof ResizeObserver === 'undefined' ? null : new ResizeObserver(measure);
        observer?.observe(el);
        window.addEventListener('resize', measure);
        return () => {
            observer?.disconnect();
            window.removeEventListener('resize', measure);
        };
    }, [ref, enabled]);
    return rect;
}
```

```tsx
// frontend/src/features/incomingCall/CallOverlay.tsx
import React, { useEffect, useLayoutEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { useTranslation } from '../../i18n';
import { interleavedZIndex, nextDialogOrder, openDialogCount, subscribeDialogStack } from '../../design/dialog/dialogStack';
import { clientLog } from '../../utils/clientLog';
import { CallAudio } from './CallAudio';
import { CallScreen } from './CallScreen';
import type { AnswerStyle } from './CallScreen';
import type { IncomingCall } from './useIncomingCall';
import { useAnchorRect } from './useAnchorRect';

interface CallOverlayProps {
    call: IncomingCall;
    /** Group colour for the halo, the eyebrow dot and the answer button. */
    accentColor: string;
    answerStyle: AnswerStyle;
    /** Service display name shown above the member's name. */
    groupLabel: string;
}

/** Where a call is drawn, decided once per call when it first shows (spec D2d decision 20). */
interface CallLayer {
    messageId: number | null;
    lifted: boolean;
    /** Opening order taken when lifted: the call sits above the dialogs opened before it. */
    order: number;
    /** Dialogs open when it first showed. */
    below: number;
}

function decideLayer(messageId: number): CallLayer {
    const below = openDialogCount();
    return below > 0
        ? { messageId, lifted: true, order: nextDialogOrder(), below }
        : { messageId, lifted: false, order: 0, below };
}

/**
 * Covers the chat area while a call is active; hosts the call audio and handles Esc/Enter.
 * A call that starts while a dialog is open is lifted: portaled over the chat area's rectangle,
 * above the dialogs that were open (interleavedZIndex) and below any opened later.
 */
export const CallOverlay: React.FC<CallOverlayProps> = ({ call, accentColor, answerStyle, groupLabel }) => {
    const { t } = useTranslation();
    const overlayRef = useRef<HTMLDivElement | null>(null);
    const anchorRef = useRef<HTMLDivElement | null>(null);
    const [now, setNow] = useState(() => Date.now());
    const { phase, answer, decline, hangUp } = call;
    const hasOverlay = phase !== 'idle' && !!call.message;
    const messageId = call.message?.id ?? null;

    // The layer is fixed for the call's life, so the audio never remounts.
    const [layer, setLayer] = useState<CallLayer>({ messageId: null, lifted: false, order: 0, below: 0 });
    if (hasOverlay && messageId !== null && layer.messageId !== messageId) setLayer(decideLayer(messageId));
    const lifted = hasOverlay && layer.lifted;

    const rect = useAnchorRect(anchorRef, lifted);
    const [zIndex, setZIndex] = useState(0);
    useLayoutEffect(() => {
        if (!lifted) return;
        const sync = () => setZIndex(interleavedZIndex(layer.order));
        sync();
        return subscribeDialogStack(sync);
    }, [lifted, layer.order]);

    useEffect(() => {
        if (layer.messageId === null) return;
        clientLog('call_layer', { phase: layer.lifted ? 'lifted' : 'in_frame', count: layer.below });
    }, [layer]);

    // Tick the talk timer while connected.
    useEffect(() => {
        if (phase !== 'connected') return;
        setNow(Date.now());
        const id = setInterval(() => setNow(Date.now()), 250);
        return () => clearInterval(id);
    }, [phase]);

    // Take keyboard focus when the call starts ringing so Enter/Esc work wherever focus was
    // (a clicked sidebar button, the timeline). Never steal it from text entry or another modal.
    useEffect(() => {
        if (phase !== 'ringing' || !hasOverlay) return;
        const overlay = overlayRef.current;
        if (!overlay) return;
        const active = document.activeElement;
        if (active instanceof HTMLElement && active !== overlay) {
            if (active.isContentEditable) return;
            if (active.closest('input, textarea, select, [contenteditable=""], [contenteditable="true"]')) return;
            const modal = active.closest('[role="dialog"], [aria-modal="true"]');
            if (modal && modal !== overlay) return;
        }
        overlay.focus({ preventScroll: true });
    }, [phase, hasOverlay]);

    // Bubble-phase window keys: modals and other handlers get the key first, and we only act
    // when nothing else has claimed it (default not prevented, focus on the call or the page).
    useEffect(() => {
        if (phase === 'idle') return;
        const onKey = (e: KeyboardEvent) => {
            if (e.key !== 'Escape' && e.key !== 'Enter') return;
            if (e.defaultPrevented || e.isComposing || e.keyCode === 229) return;
            if (e.ctrlKey || e.altKey || e.metaKey || e.shiftKey) return;
            const target = e.target instanceof Element ? e.target : null;
            if (target) {
                if (target.closest('input, textarea, select, [contenteditable=""], [contenteditable="true"]')) return;
                const inOverlay = overlayRef.current?.contains(target) ?? false;
                if (!inOverlay && target !== document.body && target !== document.documentElement) return;
                // Let a focused button activate natively (e.g. Enter on Decline).
                if (e.key === 'Enter' && target.closest('button')) return;
            }
            if (e.key === 'Escape') {
                if (phase === 'ringing') decline();
                else if (phase === 'connected') hangUp();
                else return;
            } else if (phase === 'ringing') {
                answer();
            } else {
                return;
            }
            e.preventDefault();
            e.stopPropagation();
        };
        window.addEventListener('keydown', onKey);
        return () => window.removeEventListener('keydown', onKey);
    }, [phase, answer, decline, hangUp]);

    // One visually-hidden live region that outlives the call screens: a region that mounts
    // together with its text is often not announced, so this one is always in the DOM and
    // only its text changes.
    const liveText = phase === 'ringing' ? `${call.name} ${t('call.incoming')}`
        : phase === 'ended' ? t('call.ended')
        : '';
    const liveRegion = (
        <div role="status" aria-live="polite" className="sr-only">{liveText}</div>
    );

    if (phase === 'idle' || !call.message) return liveRegion;

    const talkSeconds = call.talkStartedAt ? (now - call.talkStartedAt) / 1000 : 0;
    const placement: React.CSSProperties = lifted
        ? { zIndex, left: rect?.left ?? 0, top: rect?.top ?? 0, width: rect?.width ?? 0, height: rect?.height ?? 0 }
        : {};

    const overlay = (
        <div
            ref={overlayRef}
            role="dialog"
            aria-label={call.name}
            tabIndex={-1}
            className={lifted
                ? 'sd-call-overlay fixed overflow-hidden rounded-b-sheet outline-none'
                : 'sd-call-overlay absolute inset-0 z-30 overflow-hidden outline-none'}
            style={{ '--sd-accent': accentColor, ...placement } as React.CSSProperties}
            data-testid="call-overlay"
            data-layer={lifted ? 'lifted' : 'frame'}
        >
            <div aria-hidden className="sd-call-ambient" style={{ backgroundImage: `url("${call.photoUrl}")` }} />
            <div aria-hidden className="sd-call-vignette" />
            <div className="sd-call-stage">
                <div className="sd-call-frame">
                    <div aria-hidden className="sd-call-halo" data-phase={phase} />
                    <CallScreen
                        phase={phase}
                        name={call.name}
                        groupLabel={groupLabel}
                        photoUrl={call.photoUrl}
                        talkSeconds={talkSeconds}
                        answerStyle={answerStyle}
                        onAnswer={answer}
                        onDecline={decline}
                        onHangUp={hangUp}
                    />
                </div>
            </div>
            <CallAudio
                key={call.message.id}
                phase={phase}
                audioUrl={call.audioUrl}
                onPlaybackEnded={call.onPlaybackEnded}
                onPlaybackError={call.onPlaybackError}
            />
        </div>
    );

    return (
        <>
            {liveRegion}
            {lifted ? (
                <>
                    <div ref={anchorRef} aria-hidden="true" className="pointer-events-none absolute inset-0" />
                    {createPortal(overlay, document.body)}
                </>
            ) : overlay}
        </>
    );
};
```
